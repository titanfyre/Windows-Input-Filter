cmake_minimum_required(VERSION 3.24)
project(ProductNodeSDK LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(FAMILY_BRIDGE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../Windows-Input-Filter-C-ffi" CACHE PATH "Product C ABI source")
set(NODE_HEADERS "" CACHE PATH "Node-API headers")
set(NODE_LIBRARY "" CACHE FILEPATH "Node import library")
add_subdirectory("${FAMILY_BRIDGE_SOURCE}" bridge)
set_target_properties(WindowsInputFilterBridge PROPERTIES OUTPUT_NAME WindowsInputFilterBridge)
add_library(windows_input_filter_sdk_node MODULE native.cpp DelayLoad.cpp)
set_target_properties(windows_input_filter_sdk_node PROPERTIES PREFIX "" SUFFIX ".node")
target_include_directories(windows_input_filter_sdk_node PRIVATE "${FAMILY_BRIDGE_SOURCE}" "${NODE_HEADERS}")
target_compile_features(windows_input_filter_sdk_node PRIVATE cxx_std_17)
target_compile_definitions(windows_input_filter_sdk_node PRIVATE NAPI_VERSION=8 NODE_GYP_MODULE_NAME=windows_input_filter_sdk_node)
target_link_libraries(windows_input_filter_sdk_node PRIVATE WindowsInputFilterBridge "${NODE_LIBRARY}" delayimp)
target_link_options(windows_input_filter_sdk_node PRIVATE /DELAYLOAD:node.exe)
if(FAMILY_TEST_BACKEND)
 target_compile_definitions(windows_input_filter_sdk_node PRIVATE ICB_TEST_BACKEND)
endif()
